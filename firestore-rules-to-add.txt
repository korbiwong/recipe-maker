// Add this block to the Firestore rules of the Firebase project named in firebase-config.js
// (Firebase console > Firestore Database > Rules), inside
//   match /databases/{database}/documents { ... }
// next to the existing rules, then press Publish.
//
// It lets each signed-in account read and write only its own 食譜製造器 data.

    match /recipeMaker/{uid}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

// For Cary's friends feature (cary/), add this block too. It lets any signed-in account read the
// small public cards (name, friend code, streak, today's calories) and write only its own.

    match /caryPublic/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == uid;
    }
